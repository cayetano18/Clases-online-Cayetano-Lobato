    <script>

        const openModalBtn = document.getElementById('openModalBtn');

        const closeModalBtn = document.getElementById('closeModalBtn');

        const aboutMeModal = document.getElementById('aboutMeModal');

        const modalContent = aboutMeModal.querySelector('div');


        function openModal() {

            aboutMeModal.classList.remove('hidden');

            // Pequeño delay para que la transición de CSS se aplique correctamente

            setTimeout(() => {

                aboutMeModal.style.opacity = '1';

                modalContent.style.transform = 'scale(1)';

            }, 10);

        }


        function closeModal() {

            aboutMeModal.style.opacity = '0';

            modalContent.style.transform = 'scale(0.95)';

            setTimeout(() => {

                aboutMeModal.classList.add('hidden');

            }, 300); // Coincide con la duración de la transición

        }


        openModalBtn.addEventListener('click', openModal);

        closeModalBtn.addEventListener('click', closeModal);


        // Cerrar modal al hacer clic fuera del contenido

        aboutMeModal.addEventListener('click', (event) => {

            if (event.target === aboutMeModal) {

                closeModal();

            }

        });


        // Cerrar modal con la tecla Escape

        document.addEventListener('keydown', (event) => {

            if (event.key === 'Escape' && !aboutMeModal.classList.contains('hidden')) {

                closeModal();

            }

        });

    </script>
